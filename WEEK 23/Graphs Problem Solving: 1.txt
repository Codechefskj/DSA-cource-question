void bsf(vector<vector<pair<int,int>>> &adj , vector<int> &dist ){
deque<int> temp ;
temp.push_back(1);
dist[1]=0 ; 
while(!temp.empty()){
    int curr = temp.front();
    temp.pop_front();
    for(auto x : adj[curr]){
      int child = x.first;
      int wt = x.second ;
      if(dist[curr] + wt < dist[child]){
          dist[child]= dist[curr]+wt ; 
           if(wt==1){
            temp.push_back(child);
           }
           else{
            temp.push_front(child);
           }
      }
    }
} 
}

Q) leetcode 2316. Count Unreachable Pairs of Nodes in an Undirected Graph?
ans) 
class Solution {
public:
    int node = 0 ; 
    void dfs(int curr , vector<vector<int>> &adj , vector<bool> &vis){
     vis[curr]= true ;
     for(int i = 0 ; i< adj[curr].size(); i++){
        int next = adj[curr][i];
        if(vis[next]!= true){
            dfs(next , adj , vis);
        }
     }
     node++;
    }

    long long countPairs(int n, vector<vector<int>>& edges) {
        vector<vector<int>> adj(n+1);
        for(int i = 0 ; i< edges.size(); i++){
            adj[edges[i][0]].push_back(edges[i][1]);
            adj[edges[i][1]].push_back(edges[i][0]);
        }
        vector<int> ans ; 
        vector<bool> vis(n, false);
        for(int i = 0 ; i< n ; i++){
            if(vis[i]==false){
                dfs(i , adj , vis);
                 ans.push_back(node);
                 node= 0 ; 
            }
        } 
        long long sum = 0 ; 
        for(int x : ans) sum += x; 
        long long finalans = 0  ; 
        for(int x : ans){
        sum-=x ; 
        finalans+= sum*x * 1ll ; 
        }
        return finalans; 
    }
};

Q) leetcode 1368. Minimum Cost to Make at Least One Valid Path in a Grid?
ans) 
class Solution {
public:
    vector<pair<int,int>> movements = {
    {0,1},{0,-1},{1,0},{-1,0}
    };

    bool isvalid(int i , int j , int m , int n){
        return i>=0 && j>=0 && j<n && i<m;
    }

    void bsf(vector<vector<int>> grid , vector<vector<int>> &dist){
     int n = grid.size();
     int m = grid[0].size();
     deque<pair<int,int>> temp;
     temp.push_back({0,0});
     dist[0][0]= 0; 
     while(!temp.empty()){
       auto curr = temp.front();
       int curr_x = curr.first;
       int curr_y = curr.second;
       temp.pop_front();
       for(int i = 0 ; i< 4 ; i++){
           int child_x = movements[i].first + curr_x;
           int child_y = movements[i].second +curr_y;
           if(isvalid(child_x , child_y , n , m)){

            int cost = (grid[curr_x][curr_y]== i+1)? 0 : 1;
             
            if(dist[child_x][child_y]>dist[curr_x][curr_y]+cost){
                dist[child_x][child_y]= dist[curr_x][curr_y]+cost;
                if(cost == 0) temp.push_front({child_x,child_y});
                else{
                    temp.push_back({child_x,child_y});
                }
            }
           } 
       }
     }
    }
    int minCost(vector<vector<int>>& grid) {
        int n = grid.size();
        int m = grid[0].size();
        vector<vector<int>> dist(n, vector<int>(m, 1e9)) ; 
        bsf(grid, dist);
        return dist[n-1][m-1];
    }
};

Q) leetcode 994. Rotting Oranges?
ans) 
class Solution {
public:
    vector<pair<int,int>> movements = {
        {1,0},{0,1},{-1,0},{0,-1}
    };

    bool isvalid(int i , int j , int n , int m){
        return i<n && j<m  && i>=0 && j>=0 ;
    }

    void multisourcebsf(vector<vector<int>>& grid , vector<vector<bool>> &visited , vector<vector<int>> &dist){
        queue<pair<int,int>> temp; 

        for(int i = 0 ; i< grid.size(); i++){
            for(int j = 0 ; j< grid[0].size(); j++){
                if(grid[i][j]==2){
                    temp.push({i,j});
                    visited[i][j]= true ;
                    dist[i][j]= 0 ;
                }
            }
        }

        while(!temp.empty()){
        auto x = temp.front();
        temp.pop();
        int curr_x = x.first ; 
        int curr_y = x.second;
        for(auto mov : movements){
            int child_x = curr_x + mov.first ; 
            int child_y = curr_y + mov.second ; 
            if(isvalid(child_x , child_y , grid.size(), grid[0].size()) && visited[child_x][child_y]==false && 
            grid[child_x][child_y]==1){
                temp.push({child_x, child_y});
                visited[child_x][child_y]= true ; 
                dist[child_x][child_y]= dist[curr_x][curr_y]+1;
            }
        }
        }
    }

    int orangesRotting(vector<vector<int>>& grid) {
        vector<vector<bool>> visited(grid.size(), vector<bool>(grid[0].size(), false));
        vector<vector<int>> dist(grid.size(), vector<int>(grid[0].size(), -1));
        multisourcebsf(grid , visited , dist);
        int ans = 0 ; 
        for(int i = 0 ; i< grid.size(); i++){
            for(int j = 0 ; j< grid[0].size(); j++){
                ans = max(ans , dist[i][j]);
            }
        }
        for(int i = 0 ; i< grid.size(); i++){
            for(int j = 0 ; j< grid[0].size(); j++){
                if(grid[i][j]==1 && visited[i][j]==false){
                    return -1 ; 
                }
            }
        }
        
        return ans ; 

    }
};
