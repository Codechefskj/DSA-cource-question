Q) leetcode 1319. Number of Operations to Make Network Connected?
ans) 
class Solution {
public:
    void dfs(int curr, vector<vector<int>>& adj, vector<bool> &vis) {
        vis[curr] = true;

        for (int next : adj[curr]) {
            if (vis[next]) continue;
            dfs(next, adj, vis);
        }
    }

    int makeConnected(int n, vector<vector<int>>& connections) {
        if (connections.size() < n - 1) return -1;

        vector<vector<int>> adj(n);

        for (auto &edge : connections) {
            adj[edge[0]].push_back(edge[1]);
            adj[edge[1]].push_back(edge[0]);
        }

        vector<bool> vis(n, false);

        int components = 0;

        for (int i = 0; i < n; i++) {
            if (vis[i]) continue;
            dfs(i, adj, vis);
            components++;
        }

        return components - 1; 
    }
};

Q) 1971. Find if Path Exists in Graph
ans) 
class Solution {
public:
    vector<int> bfs(int source, vector<vector<int>>& adj, int n) {
        queue<int> q;
        vector<int> dis(n, -1);
        q.push(source);
        dis[source] = 0;
        while (!q.empty()) {
            int curr = q.front();
            q.pop();
            for (int x : adj[curr]) {
                if (dis[x] == -1) {
                    dis[x] = dis[curr] + 1;
                    q.push(x);
                }
            }
        }
        return dis;
    }

    bool validPath(int n, vector<vector<int>>& edges, int source, int destination) {
        vector<vector<int>> adj(n);
        for (auto &e : edges) {
            adj[e[0]].push_back(e[1]);
            adj[e[1]].push_back(e[0]);
        }
        vector<int> dist = bfs(source, adj, n);
        return dist[destination] != -1;  
    }
};

Q) gfg Undirected Graph Cycle?
ans) 
class Solution {
public:
    bool dfs(vector<vector<int>> &adj, vector<bool> &vis, int curr, int parent) {
        vis[curr]=true ; 
        for(int i = 0 ; i< adj[curr].size(); i++){
            int next = adj[curr][i];
            if(vis[next] && next!= parent) return true ; 
            if(next == parent) continue ; 
            if(vis[next]!=true){
                if(dfs(adj, vis , next , curr)) return true;
            }
        }
        return false ; 
    }
    
    bool isCycle(int V, vector<vector<int>>& edges) {
        vector<vector<int>> adj(V);
        for (auto &e : edges) {
            adj[e[0]].push_back(e[1]);
            adj[e[1]].push_back(e[0]);
        }
        vector<bool> vis(V, false);
        for (int i = 0; i < V; i++) {
            if (!vis[i]) {
                if (dfs(adj, vis, i, -1)) {
                    return true;
                }
            }
        }
        return false;
    }
};

Q) 733. Flood Fill ?
ans) 
class Solution {
public:
    void dfs(int i, int j, vector<vector<int>> &image, int color, int initialcolour) {
        int n = image.size();
        int m = image[0].size();
        if (i < 0 || j < 0 || i >= n || j >= m) return;

        if (image[i][j] != initialcolour) return;

        image[i][j] = color;

        dfs(i + 1, j, image, color, initialcolour);
        dfs(i - 1, j, image, color, initialcolour);
        dfs(i, j + 1, image, color, initialcolour);
        dfs(i, j - 1, image, color, initialcolour);
    }

    vector<vector<int>> floodFill(vector<vector<int>>& image, int sr, int sc, int color) {
        int initialcolour = image[sr][sc];

        if (initialcolour == color) return image;

        dfs(sr, sc, image, color, initialcolour);

        return image;
    }
};

ans) 
class Solution {
public:
    void dfs(int i , int j , vector<vector<char>> &grid , vector<vector<bool>>& vis){
    // check 
    int n = grid.size();
    int m = grid[0].size();
    if(i < 0 || j < 0 || i>=n || j>=m || grid[i][j]=='0'|| vis[i][j]) return;
    // put condition
    vis[i][j]= true ;  
     dfs(i+1 , j , grid , vis);
     dfs(i , j+1 , grid , vis);
     dfs(i-1 , j , grid , vis);
     dfs(i , j-1 , grid , vis);
    }

    int numIslands(vector<vector<char>>& grid) {
        vector<vector<bool>> vis(grid.size(), vector<bool>(grid[0].size(), false));
        int ans = 0 ; 
        for(int i = 0 ; i< grid.size(); i++){
            for(int j = 0 ; j< grid[0].size(); j++){
               if(grid[i][j]=='1' && vis[i][j]==false){
                 dfs(i , j, grid , vis);
                 ans++;
               }
            }
        }
        return ans ; 
    }
};
