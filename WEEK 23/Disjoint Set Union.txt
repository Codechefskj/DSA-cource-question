#include <bits/stdc++.h>
using namespace std;
class DSU {
public:
    vector<int> parent, size;
    DSU(int n) {
        parent.resize(n + 1);
        size.resize(n + 1, 1);
        for (int i = 0; i <= n; i++) {
            parent[i] = i;
        }
    }

    int find(int x) {
        if (parent[x] == x)
            return x;
        return parent[x] = find(parent[x]);
    }
    // union by size
    void unite(int a, int b) {
        int u = find(a);
        int v = find(b);
        if (u == v) return;
        // attach smaller tree to larger tree
        if (size[u] < size[v]) {
            swap(u, v);
        }
        parent[v] = u;
        size[u] += size[v];
    }
    bool same(int a, int b) {
        return find(a) == find(b);
    }
};

int main() {
    int n;
    cin >> n;

    DSU dsu(n);

    dsu.unite(1, 2);

    cout << dsu.same(1, 2) << " "; // 1 (true)
    cout << dsu.same(2, 3) << " "; // 0 (false)

    dsu.unite(1, 3);

    cout << dsu.same(2, 3) << " "; // 1 (true)

    return 0;
}

Q) leetcode 2368. Reachable Nodes With Restrictions ?
ans) 
class Solution {
public:
    int helper(vector<vector<int>>& adj, int curr, int parent, unordered_set<int> &restrictedSet) {
        int ans = 1; 
        for (int i = 0; i < adj[curr].size(); i++) {
            int next = adj[curr][i];
            if (next == parent) continue;
            if (restrictedSet.count(next)) continue;
            ans += helper(adj, next, curr, restrictedSet);
        }
        return ans;
    }

    int reachableNodes(int n, vector<vector<int>>& edges, vector<int>& restricted) {
        vector<vector<int>> adj(n);
        for (int i = 0; i < edges.size(); i++) {
            adj[edges[i][0]].push_back(edges[i][1]);
            adj[edges[i][1]].push_back(edges[i][0]);
        }
        unordered_set<int> restrictedSet(restricted.begin(), restricted.end());
        if (restrictedSet.count(0)) return 0;
        return helper(adj, 0, -1, restrictedSet);
    }
};

time complexity : O(N)
space complexity : O(N)

Q) leetcode 1061. Lexicographically Smallest Equivalent String ?
ans) 

#include <bits/stdc++.h>
using namespace std;

class Solution {
public:
    vector<int> parent;
    int find(int x) {
        if (parent[x] == x) return x;
        return parent[x] = find(parent[x]);
    }
    void unite(int a, int b) {
        int pa = find(a);
        int pb = find(b);
        if (pa == pb) return;
        if (pa < pb) parent[pb] = pa;
        else parent[pa] = pb;
    }

    string smallestEquivalentString(string s1, string s2, string baseStr) {
        parent.resize(26);

        for (int i = 0; i < 26; i++)
            parent[i] = i;
        for (int i = 0; i < s1.size(); i++) {
            unite(s1[i] - 'a', s2[i] - 'a');
        }

        string result = "";
        for (char c : baseStr) {
            int root = find(c - 'a');
            result += char(root + 'a');
        }

        return result;
    }
};
