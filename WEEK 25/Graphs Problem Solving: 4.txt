class Solution {
public:
    vector<pair<int,int>> movements= {
        {-1,0}, {1,0}, {0,-1},{0,1}
    };
    
    bool isvalid(int i , int j , int n , int m){
     return i >= 0 && i < n && j >= 0 && j < m;
}
    
    int dfs(vector<vector<int>>& dp , int i , int j , vector<vector<int>>& matrix , int n , int m ){
    int ans = 1 ; 
    if(dp[i][j]!= -1) return dp[i][j];
    for(int k = 0 ; k< 4 ; k++){
        auto x_direction = movements[k].first;
        auto y_direction= movements[k].second ; 
        if((isvalid(i+x_direction , j+y_direction , n , m)) && (matrix[i+x_direction][j+y_direction] > matrix[i][j])){
            ans = max(ans , dfs(dp , i+x_direction , j+y_direction , matrix  , n , m)+1);
        }
    }
    return dp[i][j]= ans ; 
    }

    int longestIncreasingPath(vector<vector<int>>& matrix) {
        int n,m ;
        n = matrix.size();
        m = matrix[0].size();
        vector<vector<int>> dp(n , vector<int>(m , -1));
        int ans = 0 ;
        for(int i = 0 ; i< n ; i++){
            for(int j = 0 ; j< m ; j++){
                ans= max(ans , dfs(dp , i , j , matrix , n ,m));
            }
        }
        return ans ; 
    }
};

Q) leetcode 1293. Shortest Path in a Grid with Obstacles Elimination ?
ans) 
class Solution {
public:

    vector<pair<int,int>> dir = {{1,0},{-1,0},{0,1},{0,-1}
    };

    int shortestPath(vector<vector<int>>& grid, int k) {
        int m = grid.size();
        int n = grid[0].size();
        vector<vector<vector<bool>>> vis(m,vector<vector<bool>>(n,vector<bool>(k + 1, false)));
        queue<vector<int>> q;
        q.push({0, 0, k});
        vis[0][0][k] = true;
        int steps = 0;
        while(!q.empty()) {
            int sz = q.size();
            while(sz--) {
                auto curr = q.front();
                q.pop();
                int i = curr[0];
                int j = curr[1];
                int rem = curr[2];
                if(i == m - 1 && j == n - 1)
                    return steps;

                for(auto d : dir) {
                    int ni = i + d.first;
                    int nj = j + d.second;
                    if(ni < 0 || nj < 0 ||
                       ni >= m || nj >= n)
                        continue;
                    if(grid[ni][nj] == 0) {
                        if(!vis[ni][nj][rem]) {
                            vis[ni][nj][rem] = true;
                            q.push({ni, nj, rem});
                        }
                    }
                    else {
                        if(rem > 0 &&
                           !vis[ni][nj][rem - 1]) {
                            vis[ni][nj][rem - 1] = true;
                            q.push({ni, nj, rem - 1});
                        }
                    }
                }
            }
            steps++;
        }
        return -1;
    }
};

Q) 947. Most Stones Removed with Same Row or Column ?
ans) 
class Solution {
public:
    void dfs(int node , vector<vector<int>>& stones , vector<bool>& visited){
    visited[node]= true ; 
    for(int nxt = 0 ; nxt<stones.size() ; nxt++){
        if((visited[nxt]==false) &&(stones[node][0]==stones[nxt][0] ||stones[node][1]==stones[nxt][1])){
           dfs(nxt, stones, visited);
     }
    }
    }

    int removeStones(vector<vector<int>>& stones) {
        int n = stones.size();
        int components = 0 ; 
        vector<bool> visited(n , false);
        for(int i = 0 ; i< n ; i++){
            if(visited[i]==false){
                components++;
                dfs(i,stones , visited);
            }
        }
        return n- components ; 
    }
};
