strong connected components :
In a directed graph, a Strongly Connected Component is a group of vertices where:
Every vertex can reach every other vertex.
Path must follow the direction of edges.

directed acyclic graph : 
where edges have direction
(Directed)
and there is no cycle
(Acyclic)

type of edges ->

tree edges: An edge used to visit a node for the first time.

back edges: An edge from a node to its ancestor in DFS tree.

forward edges : An edge from a node to its descendant but not a tree edge.

cross edges : An edge connecting two unrelated subtrees/nodes.

class Solution {
public:
    bool iscycle(int curr,vector<bool>& visited,vector<bool>& isrecursion,vector<vector<int>>& adj){
        visited[curr] = true;
        isrecursion[curr] = true;
        for(auto x : adj[curr]){
            if(visited[x] && isrecursion[x]){
                return true;
            }
            if(!visited[x]){
                if(iscycle(x, visited, isrecursion, adj)){
                    return true;
                }
            }
        }
        isrecursion[curr] = false;
        return false;
    }

    void findans(int curr,vector<int>& visitedd,vector<int>& ans,vector<vector<int>>& adj){
        visitedd[curr] = 0;
        for(auto x : adj[curr]){
            if(visitedd[x] == -1){
                findans(x, visitedd, ans, adj);
            }
        }
        ans.push_back(curr);
    }

    vector<int> findOrder(int n,vector<vector<int>>& edges) {
        vector<vector<int>> adj(n);
        for(int i = 0; i < edges.size(); i++){
            int u = edges[i][0];
            int v = edges[i][1];
            adj[v].push_back(u);
        }
        vector<int> visitedd(n, -1);
        vector<int> ans;

        vector<bool> isrecursion(n, false);
        vector<bool> visited(n, false);
        bool iscycleornot = false;
        for(int i = 0; i < n; i++){
            if(!visited[i]){
                iscycleornot |= iscycle(i,visited,isrecursion,adj);
            }
        }
        for(int i = 0; i < n; i++){ 
            if(visitedd[i] == -1){
                findans(i,visitedd,ans,adj);
            }
        }

        reverse(ans.begin(), ans.end());

        return (!iscycleornot) ? ans : vector<int>{};
    }
};

class Solution {
public:
    bool findans(int src , int des, vector<bool>& visited , vector<vector<int>>& adj){
    visited[src]= true ; 
    if(src==des) return true ; 
    bool ans = false ; 
    for(auto x : adj[src]){
        if(visited[x]==false){
           if(findans(x , des , visited , adj)){
            ans = true ; 
           }
        }
    }
    return ans ;
    }

    vector<bool> checkIfPrerequisite(int n, vector<vector<int>>& edges, vector<vector<int>>& queries) {
        vector<vector<int>> adj(n);
        for(auto x : edges){
         int u = x[0];
         int v = x[1];
         adj[u].push_back(v);
        }
        vector<bool> ans(queries.size());
        vector<bool> visited(n , false);
        for(int i = 0 ; i<queries.size(); i++){
            int u = queries[i][0];
            int v = queries[i][1];
            visited.assign(n , false);
            if(findans(u , v , visited , adj)){
                ans[i]= true; 
            }
        }
        return ans ; 
    }
};

ans) 
// we check cycle exists then we dont count that path because its is not a safe path 
class Solution {
public:

    bool iscycleisnot(int curr,vector<vector<int>>& adj,vector<bool>& visited,vector<bool>& isrecursion) {
        visited[curr] = true;
        isrecursion[curr] = true;
        for (auto x : adj[curr]) {
            if (visited[x] && isrecursion[x]) {
                return true;
            }
            if (!visited[x]) {
                if (iscycleisnot(x, adj, visited, isrecursion)) {
                    return true;
                }
            }
        }
        isrecursion[curr] = false;
        return false;
    }

    vector<int> eventualSafeNodes(vector<vector<int>>& graph) {

        int n = graph.size();

        vector<int> ans;

        for (int i = 0; i < n; i++) {

            vector<bool> visited(n, false);
            vector<bool> isrecursion(n, false);

            if (!iscycleisnot(i, graph, visited, isrecursion)) {
                ans.push_back(i);
            }
        }
        sort(ans.begin(), ans.end());

        return ans;
    }
};
