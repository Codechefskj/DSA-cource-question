there are two ways to represent the tree / graph 1) adjacent matrix 2) adjacent list 
we usually used adjacent list for storing the list 
in adjacent matrix : O(n^2) - space complexity which is very large and N!= 10^5 
                     we only allowed N<=10^3 

in adjacent list : O(N+M) space complexity 
                N= 10^5 , E= 10^7 maximum 

but if we find the weight of graph we spent lot of time for calculate in adjacent list but spent less time to find out of 
weight in adjacent matrix

 #include <bits/stdc++.h>
using namespace std;

void bfs(vector<vector<int>> &adj, int start) {
    vector<bool> vis(adj.size(), false);
    queue<int> q;
    q.push(start);
    vis[start] = true;
    while (!q.empty()) {
        int curr = q.front();
        q.pop();
        cout << curr << " ";
        for (int i = 0; i < adj[curr].size(); i++) {
            int next = adj[curr][i];
            if (!vis[next]) {
                vis[next] = true;
                q.push(next);
            }
        }
    }
}

Q) leetcode 1443. Minimum Time to Collect All Apples in a Tree ?
ans) 
class Solution {
public:
    int dfs(int curr , int parents ,vector<vector<int>>& adj, vector<bool>& hasapple){
    int ans = 0 ; 
    for(int i = 0 ; i< adj[curr].size() ; i++){
        int next = adj[curr][i];
        if(next!=parents){
           int x = dfs(next , curr , adj , hasapple);
           if(x>0 || hasapple[next]){
            ans+= x+2;
           }
        }
    }
    return ans ;
    }

    int minTime(int n, vector<vector<int>>& edges, vector<bool>& hasApple) {
     vector<vector<int>> adj(n+1);
     for(int i = 0 ; i < edges.size() ; i++){
            adj[edges[i][0]].push_back(edges[i][1]);
            adj[edges[i][1]].push_back(edges[i][0]);
     }
     return dfs(0 , -1 ,adj , hasApple );
    }
};
